57/// Entry point of the application.
58///
59/// # Returns
60///
61/// Returns `Ok(())` if successful, or an error if an operation fails.
62pub fn run() -> Result<(), Box<dyn Error>> {
63    // Initialize the logger using the `env_logger` crate
64    init_logger(None)?;
65
66    // A clap error here is `--help`, `--version`, or a usage mistake.
67    // `Error::exit` renders it the way a CLI should: help and version to
68    // stdout with status 0, usage errors to stderr with status 2.
69    // `run_with` returns the error instead of exiting, so that it stays
70    // callable from a test without killing the test process — which is
71    // the same split as `run_cli` and `run_cli_from`.
72    match run_with(std::env::args_os()) {
73        Err(e) => match e.downcast::<clap::Error>() {
74            Ok(clap_error) => clap_error.exit(),
75            Err(other) => Err(other),
76        },
77        ok => ok,
78    }
79}

81/// Runs the application against an explicit argument list.
82///
83/// [`run`] delegates here after initialising the logger. Split out for
84/// the same reason [`cli::run_cli_from`] exists: `run` reads
85/// `std::env::args_os()`, which under a test harness holds the
86/// harness's own arguments, so the whole body was unreachable from the
87/// suite and sat at zero coverage.
88///
89/// # Errors
90///
91/// Returns an error if the log directory or file cannot be created, if
92/// the arguments fail to parse, or if generating the quote fails.
93pub fn run_with<I, T>(args: I) -> Result<(), Box<dyn Error>>
94where
95    I: IntoIterator<Item = T>,
96    T: Into<std::ffi::OsString> + Clone,
97{
98    // Define date and time
99    let date = DateTime::new();
100    let iso = date.format_rfc3339()?;
101
102    // Ensure log directory exists and open log file
103    let log_dir = Path::new(OUTPUT_DIR).join("logs");
104    fs::create_dir_all(&log_dir)?;
105    let log_path = log_dir.join("wiserone.log");
106    let mut log_file = File::create(&log_path)?;
107
108    // Call into the CLI with the supplied arguments
109    cli::run_cli_from(args)?;
110
111    // Generate a log entry
112    let quote_log = macro_log!(
113        "id",
114        &iso,
115        &LogLevel::INFO,
116        "process",
117        "Quote HTML file generated successfully.",
118        &LogFormat::CLF
119    );
120
121    // Write the log to both the console and the file
122    writeln!(log_file, "{}", quote_log)?;
123
124    Ok(())
125}
