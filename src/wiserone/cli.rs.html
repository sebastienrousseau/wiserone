21/// The directory where output files (including logs) are stored.
22const OUTPUT_DIR: &str = "./docs";
23
24#[derive(Parser)]
25#[clap(author, version, about)]
26/// A command line program that generates an HTML file containing a
27/// quote from the JSON file. The program can generate a random quote
28/// or all quotes from the JSON file.
29
30#[derive(Debug)]
31pub enum Command {
32    /// Selects a random quote from the JSON or CSV file and creates an HTML
33    /// file based on the quote.
34    Random {
35        /// The name of the JSON or CSV file containing quotes.
36        filename: String,
37    },
38    /// Selects the quote of the day — the same one wiserone.com shows —
39    /// and creates an HTML file based on it.
40    Daily {
41        /// The name of the JSON or CSV file containing quotes.
42        filename: String,
43    },
44    /// Selects all quotes from the JSON or CSV file and creates an HTML file
45    /// for each quote.
46    All {
47        /// The name of the JSON file containing quotes.
48        filename: String,
49    },
50}
51
52/// The entry point of the program.
53///
54/// # Arguments
55///
56/// * `args`: The command line arguments passed to the program.
57///
58/// # Returns
59///
60/// * `i32`: An exit code indicating the success or failure of the
61///   program.
62pub fn run_cli() -> Result<(), Box<dyn Error>> {
63    match run_cli_from(std::env::args_os()) {
64        // A clap error here is `--help`, `--version`, or a usage
65        // mistake. `Error::exit` renders it the way a CLI should:
66        // help and version to stdout with status 0, usage errors to
67        // stderr with status 2. Returning it instead would make
68        // `wiserone --help` exit non-zero.
69        Err(e) => match e.downcast::<clap::Error>() {
70            Ok(clap_error) => clap_error.exit(),
71            Err(other) => Err(other),
72        },
73        ok => ok,
74    }
75}
76
77/// Runs the CLI against an explicit argument list.
78///
79/// [`run_cli`] delegates here with [`std::env::args_os`]. Prefer this
80/// wherever the arguments should not depend on how the process was
81/// started: under `cargo bench` the process arguments include
82/// `--bench`, and under `cargo test` they belong to the test harness,
83/// so parsing them would fail for reasons unrelated to this CLI.
84///
85/// # Errors
86///
87/// Returns an error if `args` fail to parse, or if generating the
88/// output files fails.
89pub fn run_cli_from<I, T>(args: I) -> Result<(), Box<dyn Error>>
90where
91    I: IntoIterator<Item = T>,
92    T: Into<std::ffi::OsString> + Clone,
93{
94    // Ensure log directory exists and open log file
95    let log_dir = Path::new(OUTPUT_DIR).join("logs");
96    fs::create_dir_all(&log_dir)?;
97    let log_path = log_dir.join("wiserone.log");
98    let mut log_file = File::create(&log_path)?;
99
100    // Define date and time
101    let dt = DateTime::new();
102    let iso = dt.format_rfc3339()?;
103    let year = dt.year();
104    let month = &iso[5..7];
105    let day = dt.day();
106    let date = format!("{}_{}_{}", year, month, day);
107
108    // Generate a log entry
109    let ascii_art_log = macro_log!(
110        "id",
111        &iso,
112        &LogLevel::INFO,
113        "process",
114        "ASCII art generation event started.",
115        &LogFormat::CLF
116    );
117
118    match generate_ascii_art("The Wiser One") {
119        Ok(ascii_art) => println!("{}", ascii_art),
120        Err(e) => eprintln!("Error generating ASCII art: {:?}", e),
121    }
122
123    // Write the log to both the console and the file
124    writeln!(log_file, "{}", ascii_art_log)?;
125
126    // Parse the command line arguments using the `clap` crate.
127    let command = Command::try_parse_from(args)?;
128
129    match command {
130        Command::Random { filename } => {
131            println!(
132                "- info:wiserone: begin generating a random quote"
133            );
134            // Construct the HTML filename using `iso`
135            let html_filename = format!("{}.html", date);
136
137            // Read and parse quotes, then select a random quote
138            let mut quotes = read_quotes_from_file(&filename)?;
139            let quote = quotes.select_random_quote()?;
140            generate_html_file(&html_filename, quote)?;
141            generate_sitemap_file("https://wiserone.com/")?;
142        }
143        Command::Daily { filename } => {
144            println!(
145                "- info:wiserone: begin generating the quote of the day"
146            );
147            let html_filename = format!("{}.html", date);
148            let quotes = read_quotes_from_file(&filename)?;
149            let quote =
150                quotes.select_daily_quote(current_day_number())?;
151            generate_html_file(&html_filename, quote)?;
152            generate_sitemap_file("https://wiserone.com/")?;
153        }
154        Command::All { filename } => {
155            println!("- info:wiserone: begin generating all quotes");
156            // Read and parse all quotes
157            let quotes = read_quotes_from_file(&filename)?;
158
159            // Generate an HTML file for each quote
160            for quote in quotes.select_all_quotes()? {
161                // Name by pool position, not by date. `date_added` used
162                // to be one-per-day and unique; in a pool it records the
163                // day a line was written, and six of the current 136
164                // share a date. Naming by it silently overwrote files.
165                let html_filename = match quote.id {
166                    Some(id) => format!("quote-{:04}.html", id),
167                    None => {
168                        let date_part = quote
169                            .date_added
170                            .split('T')
171                            .next()
172                            .unwrap_or("");
173                        format!("{}.html", date_part.replace('-', "_"))
174                    }
175                };
176                generate_html_file(&html_filename, quote)?;
177                generate_sitemap_file("https://wiserone.com/")?;
178            }
179            println!("- info:wiserone: end generating all quotes\n\n");
180        }
181    }
182
183    Ok(())
184}
