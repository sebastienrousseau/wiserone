5use crate::quotes::Quote;
6use dtt::datetime::DateTime;
7use rlg::log_format::LogFormat;
8use rlg::log_level::LogLevel;
9use rlg::macro_log;
10use std::fmt::format;
11use std::{
12    error::Error,
13    fs::{self, File},
14    io::Write,
15    path::Path,
16};
17use uuid::Uuid;
18
19/// Creates an HTML file based on the provided quote.
20///
21/// # Arguments
22///
23/// * `filename` - The name of the file to be created.
24/// * `quote` - A reference to the quote to be used.
25///
26/// # Returns
27///
28/// Returns `Ok(())` if the file is successfully created, or an error
29/// otherwise.
30pub fn generate_html_file(
31    filename: &str,
32    quote: &Quote,
33) -> Result<(), Box<dyn Error>> {
34    let mut layout = fs::read_to_string("_layouts/quote.html")?;
35
36    // Define date and time
37    let dt = DateTime::new();
38    let iso = dt.format_rfc3339()?;
39    let year = dt.year();
40    let month = &iso[5..7];
41    let day = dt.day();
42
43    // Determine if the date matches today
44    let is_today = year == dt.year()
45        && month == {
46            let res =
47                format(format_args!("{:02}", u8::from(dt.month())));
48            res
49        }
50        && day == dt.day();
51
52    let date = format!("{}_{}_{}", year, month, day);
53    let prefix = if is_today {
54        "https://wiserone.com/index.html".to_string() // If the date is today
55    } else {
56        format!("https://wiserone.com/{}.html", date) // For any other date
57    };
58
59    println!("Prefix: {}", prefix);
60
61    // Replace the placeholders with values from the quote
62    layout = layout.replace("{{apple_touch_icon_sizes}}", "192x192");
63    layout = layout.replace("{{author}}", &quote.author);
64    layout = layout.replace("{{banner}}", &quote.image_url);
65    layout = layout.replace("{{cdn}}", "https://cloudcdn.pro");
66    layout = layout.replace("{{charset}}", "utf-8");
67    layout = layout.replace("{{description}}", "Daily nuggets of wisdom in a clean, minimalist design, inspiring deeper thought and personal growth with every visit.");
68    layout = layout.replace("{{hreflang}}", "en");
69    layout = layout.replace("{{item_pub_date}}", &quote.date_added);
70    layout = layout.replace(
71        "{{date}}",
72        quote.date_added.split('T').next().unwrap_or(""),
73    );
74    layout = layout.replace(
75        "{{logo}}",
76        "https://cloudcdn.pro/clients/wiserone/v1/logos/wiserone.svg",
77    );
78    layout = layout.replace("{{measurementID}}", "G-4HKZ6N3QSC");
79    layout = layout.replace("{{name}}", "wiserone");
80    layout = layout.replace("{{title}}", &quote.quote_text);
81    layout = layout.replace("{{url}}", "https://wiserone.com");
82    layout = layout.replace("{{canonical}}", &prefix);
83
84    fs::create_dir_all("./docs")?;
85    let path = Path::new("./docs").join(filename);
86    let mut file = fs::File::create(&path)?;
87    file.write_all(layout.as_bytes())?;
88
89    // Open the log file for appending
90    let mut log_file = File::create("./wiserone.log")?;
91
92    // Collect filenames into a vector, exclude .DS_Store, and sort them alphabetically
93    let mut filenames: Vec<_> = fs::read_dir("./docs")?
94        .filter_map(|entry| {
95            entry.ok().map(|e| {
96                let path = e.path();
97                let path_str = path.to_string_lossy().into_owned();
98                path_str
99            })
100        })
101        .filter(|filename| !filename.ends_with(".DS_Store"))
102        .collect();
103
104    filenames.sort(); // Sort filenames alphabetically
105
106    // Iterate over sorted filenames and log each one
107    for filename in &filenames {
108        let uuid = Uuid::new_v4();
109
110        // Write the log to both the console and the file
111        let file_log = macro_log!(
112            &uuid.to_string(),
113            &iso,
114            &LogLevel::INFO,
115            "process",
116            &format!("The HTML File is created at `{}`.", filename),
117            &LogFormat::CLF
118        );
119        writeln!(log_file, "{}", file_log)?;
120
121        // Assuming year, month, and day are already defined correctly
122        let today_formatted = format!(
123            "{year}_{month:02}_{day:02}",
124            year = year,
125            month = month,
126            day = day
127        );
128
129        // Create the file path for the current day's file if it doesn't already exist
130        let today_file_path =
131            format!("./docs/{}.html", today_formatted);
132
133        if Path::new(&today_file_path).exists() {
134            let content = fs::read_to_string(&today_file_path)?;
135            let index_path = Path::new("./docs/index.html");
136            fs::write(index_path, content.as_bytes())?;
137
138            // Write the log to both the console and the file
139            let file_log = macro_log!(
140                &Uuid::new_v4().to_string(),
141                &iso,
142                &LogLevel::INFO,
143                "process",
144                &format!(
145                    "index.html updated with content from {}",
146                    today_file_path
147                ),
148                &LogFormat::CLF
149            );
150            writeln!(log_file, "{}", file_log)?;
151        } else {
152            // Write the log to both the console and the file
153            let file_log = macro_log!(
154                &Uuid::new_v4().to_string(),
155                &iso,
156                &LogLevel::INFO,
157                "process",
158                &format!("No file found at {}", today_file_path),
159                &LogFormat::CLF
160            );
161            writeln!(log_file, "{}", file_log)?;
162        }
163    }
164    println!("- info:wiserone: add file at `{}`", path.display());
165    Ok(())
166}
