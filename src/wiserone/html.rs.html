17/// The directory where HTML files are generated.
18const OUTPUT_DIR: &str = "./docs";
19
20/// The path to the HTML template file.
21const TEMPLATE_PATH: &str = "_layouts/quote.html";
22
23/// Validates that a filename is safe for use in file operations.
24///
25/// # Arguments
26///
27/// * `filename` - The filename to validate.
28///
29/// # Returns
30///
31/// Returns `Ok(())` if the filename is safe, or an error if unsafe.
32fn validate_filename(filename: &str) -> Result<(), Box<dyn Error>> {
33    // Check for directory traversal sequences
34    if filename.contains("..")
35        || filename.contains('/')
36        || filename.contains('\\')
37    {
38        return Err(
39            "Invalid filename: contains directory traversal characters"
40                .into(),
41        );
42    }
43
44    // Check for valid HTML extension
45    if !filename.ends_with(".html") {
46        return Err("Invalid filename: must end with .html".into());
47    }
48
49    // Check for empty or whitespace-only names
50    let name_without_ext = filename.trim_end_matches(".html");
51    if name_without_ext.is_empty()
52        || name_without_ext.chars().all(|c| c.is_whitespace())
53    {
54        return Err("Invalid filename: name cannot be empty".into());
55    }
56
57    Ok(())
58}

60/// Validates that the template file exists and is readable.
61///
62/// # Returns
63///
64/// Returns `Ok(())` if the template is valid, or an error otherwise.
65fn validate_template() -> Result<(), Box<dyn Error>> {
66    let template_path = Path::new(TEMPLATE_PATH);
67
68    if !template_path.exists() {
69        return Err(format!(
70            "Template file not found: {}",
71            TEMPLATE_PATH
72        )
73        .into());
74    }
75
76    if !template_path.is_file() {
77        return Err(format!(
78            "Template path is not a file: {}",
79            TEMPLATE_PATH
80        )
81        .into());
82    }
83
84    Ok(())
85}

87/// Creates an HTML file based on the provided quote.
88///
89/// # Arguments
90///
91/// * `filename` - The name of the file to be created (must be a simple filename, not a path).
92/// * `quote` - A reference to the quote to be used.
93///
94/// # Returns
95///
96/// Returns `Ok(())` if the file is successfully created, or an error
97/// otherwise.
98///
99/// # Security
100///
101/// This function validates the filename to prevent directory traversal attacks.
102/// Files are always created in the designated output directory (./docs).
103pub fn generate_html_file(
104    filename: &str,
105    quote: &Quote,
106) -> Result<(), Box<dyn Error>> {
107    generate_html_file_in(filename, quote, Path::new(OUTPUT_DIR))
108}

110/// Generates an HTML file for `quote` inside `output_dir`.
111///
112/// Behaves exactly like [`generate_html_file`], but writes into the
113/// directory given rather than the default `./docs`. Prefer this in
114/// tests so a run never touches the project's own output tree.
115///
116/// # Arguments
117///
118/// * `filename` - name of the file to create inside `output_dir`.
119/// * `quote` - the quote rendered into the template.
120/// * `output_dir` - directory written into; created if absent.
121///
122/// # Errors
123///
124/// Returns an error if the filename fails validation, the template is
125/// missing, or the directory or file cannot be written.
126///
127/// # Security
128///
129/// The filename is validated to prevent directory traversal; files are
130/// always created inside `output_dir`.
131pub fn generate_html_file_in(
132    filename: &str,
133    quote: &Quote,
134    output_dir: &Path,
135) -> Result<(), Box<dyn Error>> {
136    // Validate filename to prevent path traversal
137    validate_filename(filename)?;
138
139    // Validate template exists before reading
140    validate_template()?;
141
142    let mut layout = fs::read_to_string(TEMPLATE_PATH)?;
143
144    // Define date and time
145    let dt = DateTime::new();
146    let iso = dt.format_rfc3339()?;
147    let year = dt.year();
148    let month = &iso[5..7];
149    let day = dt.day();
150
151    // The canonical is the quote's own page on wiserone.com.
152    //
153    // This used to be `if is_today { index.html } else { <date>.html }`,
154    // which was wrong twice over. The condition compared `dt` against
155    // itself — `year == dt.year() && ...` — so it was always true and the
156    // else branch was unreachable. And the URL it built,
157    // `wiserone.com/YYYY_MM_DD.html`, has never been a page the site
158    // serves; dated URLs use hyphens and, since the corpus became a
159    // pool, they all canonicalise to `/q/<slug>/` anyway.
160    let prefix =
161        format!("https://wiserone.com/q/{}/", slug(&quote.quote_text));
162
163    println!("Prefix: {}", prefix);
164
165    // Replace the placeholders with values from the quote
166    layout = layout.replace("{{apple_touch_icon_sizes}}", "192x192");
167    layout = layout.replace("{{author}}", &quote.author);
168    layout = layout.replace("{{banner}}", &quote.image_url);
169    layout = layout.replace("{{cdn}}", "https://cloudcdn.pro");
170    layout = layout.replace("{{charset}}", "utf-8");
171    layout = layout.replace("{{description}}", "Daily nuggets of wisdom in a clean, minimalist design, inspiring deeper thought and personal growth with every visit.");
172    layout = layout.replace("{{hreflang}}", "en");
173    layout = layout.replace("{{item_pub_date}}", &quote.date_added);
174    layout = layout.replace(
175        "{{date}}",
176        quote.date_added.split('T').next().unwrap_or(""),
177    );
178    layout = layout.replace(
179        "{{logo}}",
180        "https://cloudcdn.pro/clients/wiserone/v1/logos/wiserone.svg",
181    );
182    layout = layout.replace("{{measurementID}}", "G-4HKZ6N3QSC");
183    layout = layout.replace("{{name}}", "wiserone");
184    layout = layout.replace("{{title}}", &quote.quote_text);
185    layout = layout.replace("{{url}}", "https://wiserone.com");
186    layout = layout.replace("{{canonical}}", &prefix);
187
188    fs::create_dir_all(output_dir)?;
189    let path = output_dir.join(filename);
190    let mut file = File::create(&path)?;
191    file.write_all(layout.as_bytes())?;
192
193    // Ensure log directory exists and open log file
194    let log_dir = output_dir.join("logs");
195    fs::create_dir_all(&log_dir)?;
196    let log_path = log_dir.join("wiserone.log");
197    let mut log_file = File::create(&log_path)?;
198
199    // Collect filenames into a vector, exclude .DS_Store, and sort them alphabetically
200    let mut filenames: Vec<_> = fs::read_dir(output_dir)?
201        .filter_map(|entry| {
202            entry.ok().map(|e| {
203                let path = e.path();
204                let path_str = path.to_string_lossy().into_owned();
205                path_str
206            })
207        })
208        .filter(|filename| !filename.ends_with(".DS_Store"))
209        .collect();
210
211    filenames.sort(); // Sort filenames alphabetically
212
213    // Iterate over sorted filenames and log each one
214    for filename in &filenames {
215        // Write the log to both the console and the file
216        let msg =
217            format!("The HTML File is created at `{}`.", filename);
218        let file_log = Log::build(LogLevel::INFO, &msg)
219            .time(&iso)
220            .component("process")
221            .format(LogFormat::CLF);
222        writeln!(log_file, "{}", file_log)?;
223
224        // Assuming year, month, and day are already defined correctly
225        let today_formatted = format!(
226            "{year}_{month:02}_{day:02}",
227            year = year,
228            month = month,
229            day = day
230        );
231
232        // Create the file path for the current day's file if it doesn't already exist
233        let today_file_path =
234            output_dir.join(format!("{}.html", today_formatted));
235
236        if today_file_path.exists() {
237            let content = fs::read_to_string(&today_file_path)?;
238            let index_path = output_dir.join("index.html");
239            fs::write(index_path, content.as_bytes())?;
240
241            // Write the log to both the console and the file
242            let msg = format!(
243                "index.html updated with content from {}",
244                today_file_path.display()
245            );
246            let file_log = Log::build(LogLevel::INFO, &msg)
247                .time(&iso)
248                .component("process")
249                .format(LogFormat::CLF);
250            writeln!(log_file, "{}", file_log)?;
251        } else {
252            // Write the log to both the console and the file
253            let msg = format!(
254                "No file found at {}",
255                today_file_path.display()
256            );
257            let file_log = Log::build(LogLevel::INFO, &msg)
258                .time(&iso)
259                .component("process")
260                .format(LogFormat::CLF);
261            writeln!(log_file, "{}", file_log)?;
262        }
263    }
264    println!("- info:wiserone: add file at `{}`", path.display());
265    Ok(())
266}
