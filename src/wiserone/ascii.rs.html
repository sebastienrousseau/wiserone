5use figlet_rs::FIGlet;
6use std::error::Error;
7use std::fmt;
8
9/// Error type for ASCII art generation failures.
10#[derive(Debug, Clone, Copy, PartialEq, Eq)]
11pub enum ArtError {
12    /// Represents a failure to load the FIGlet.
13    FontLoadError,
14    /// Represents a failure to convert text to ASCII art.
15    ConversionError,
16}
17
18impl fmt::Display for ArtError {
19    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
20        match *self {
21            Self::FontLoadError => {
22                write!(f, "Failed to load FIGlet")
23            }
24            Self::ConversionError => {
25                write!(f, "Failed to convert text to ASCII art")
26            }
27        }
28    }
29}
30
31impl Error for ArtError {}
32
33/// Generates ASCII art from the given text using the standard `FIGlet`.
34///
35/// # Arguments
36///
37/// * `text` - The text to convert to ASCII art.
38///
39/// # Errors
40///
41/// This function returns an `Err` in the following situations:
42///
43/// - If the standard `FIGlet` fails to load (`FontLoadError`).
44/// - If the text cannot be converted to ASCII art (`ConversionError`).
45///
46pub fn generate_ascii_art(text: &str) -> Result<String, ArtError> {
47    let standard_font =
48        FIGlet::standard().map_err(|_| ArtError::FontLoadError)?;
49    let figure =
50        standard_font.convert(text).ok_or(ArtError::ConversionError)?;
51    Ok(figure.to_string())
52}
