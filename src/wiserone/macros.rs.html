7/// This macro creates a Quote struct from the provided fields.
8///
9/// # Example
10///
11/// ```
12/// use wiserone::wiserone;
13///
14/// let quote = wiserone! {
15///     quote_text: "The only way to do great work is to love what you do.",
16///     author: "Steve Jobs",
17///     date_added: "2024-01-01T00:00:00Z",
18///     image_url: "https://example.com/image.jpg"
19/// };
20/// ```
21#[macro_export]
22macro_rules! wiserone {
23    (
24        quote_text: $quote_text:expr,
25        author: $author:expr,
26        date_added: $date_added:expr,
27        image_url: $image_url:expr $(,)?
28    ) => {
29        $crate::quotes::Quote {
30            quote_text: $quote_text.to_string(),
31            author: $author.to_string(),
32            date_added: $date_added.to_string(),
33            image_url: $image_url.to_string(),
34        }
35    };
36}

46/// This macro creates a new vector of the given elements.
47#[macro_export]
48macro_rules! wiserone_vec {
49    ($($elem:expr),*) => {{
50        let mut v = Vec::new();
51        $(v.push($elem);)*
52        v
53    }};

56/// This macro creates a new map of the given key-value pairs.
57#[macro_export]
58macro_rules! wiserone_map {
59    ($($key:expr => $value:expr),*) => {{
60        use std::collections::HashMap;
61        let mut m = HashMap::new();
62        $(m.insert($key, $value);)*
63        m
64    }};

67/// This macro checks if the given expression is true.
68///
69/// # Example
70///
71/// ```
72/// use wiserone::wiserone_assert;
73///
74/// wiserone_assert!(true);
75/// wiserone_assert!(1 + 1 == 2);
76/// wiserone_assert!(vec![1, 2, 3].len() == 3);
77/// ```
78#[macro_export]
79macro_rules! wiserone_assert {
80    ($cond:expr) => {
81        if !$cond {
82            panic!("Assertion failed: {}", stringify!($cond));
83        }
84    };
85    ($cond:expr, $($msg:tt)+) => {
86        if !$cond {
87            panic!("Assertion failed: {}: {}", stringify!($cond), format!($($msg)+));
88        }
89    };
90}

92/// This macro returns the minimum of the given values.
93///
94/// # Example
95///
96/// ```
97/// use wiserone::wiserone_min;
98///
99/// assert_eq!(wiserone_min!(5), 5);
100/// assert_eq!(wiserone_min!(5, 3, 8), 3);
101/// assert_eq!(wiserone_min!(1, 2, 3, 4, 5), 1);
102/// ```
103#[macro_export]
104macro_rules! wiserone_min {
105    ($x:expr) => {
106        $x
107    };
108    ($x:expr, $($rest:expr),+) => {{
109        let first = $x;
110        let rest = wiserone_min!($($rest),+);
111        if first < rest { first } else { rest }
112    }};
113}

129        $x
130    };
131    ($x:expr, $($rest:expr),+) => {{
132        let first = $x;
133        let rest = wiserone_max!($($rest),+);
134        if first > rest { first } else { rest }
135    }};

138/// This macro takes a string and splits it into a vector of words.
139#[macro_export]
140macro_rules! wiserone_split {
141    ($s:expr) => {{
142        let mut v = Vec::new();
143        for w in $s.split_whitespace() {
144            v.push(w.to_string());
145        }
146        v
147    }};
148}

150/// This macro takes a vector of strings and joins them together into a
151/// single string.
152#[macro_export]
153macro_rules! wiserone_join {
154    ($($s:expr),*) => {{
155        let mut s = String::new();
156        $(
157            s += &$s;
158        )*
159        s
160    }};
161}
